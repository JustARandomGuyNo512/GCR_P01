Barrel aim and reticle — code to port
======================================
This is the code added for first-person barrel firing and the HUD reticle.
Copy the two classes, then add the hooks. Names that belong to this mod are
marked. Swap those for the other mod's gun, packet, muzzle bone, and matrices.

Behavior this code implements:
- Shots spawn at the muzzle and travel along the barrel. They are not bent
  back to screen center.
- The center dot is the camera. The hollow ring is the barrel ray.
- The ring turns amber when the barrel hits something the camera ray misses.
- Aiming down sights hides the whole HUD, including iron sights, once aim
  progress passes 0.45. The shot still uses the barrel while the HUD is hidden.
- The viewmodel already contains lag, sway, and recoil. Do not add a second spring.
- Spread stays server-side, around the barrel direction. The ring is the cone center.
- Projectile drop is not drawn.

A distributed port has to follow that license. It does not include the
third-party models and textures.


1. New file: client aim solver
------------------------------
Save as your own client-only class. Replace GunFirePacket, Client.WEAPON_STATUS,
Client.isUsingIrisShader, and RenderEvents.currentFov.

/**
 * First-person shot solution taken from the MUZZLE_FLASH bone.
 *
 * The camera and the barrel are separate. The center dot is where the camera looks. The
 * tethered ring is the barrel ray: shots leave the muzzle along that direction, and the ring
 * sits on the first surface that ray hits. The viewmodel already carries the gun's lag, sway,
 * and recoil, so the ring is a readout of that muzzle rather than a second aim spring.
 */
@OnlyIn(Dist.CLIENT)
public final class BarrelAim {
    private static final double BLOCK_RANGE = 192.0;
    private static final double ENTITY_RANGE = 64.0;
    /** Used when the barrel ray hits nothing, so the ring still shows barrel direction. */
    private static final double SKY_RANGE = 48.0;
    private static final long STALE_NS = 250_000_000L;

    private static final Vector3f localPos = new Vector3f();
    private static final Vector3f localDir = new Vector3f();
    private static boolean captured;
    private static long capturedAt;
    /** Most negative view-space Z wins, so a suppressor tip beats the barrel behind it. */
    private static float bestForwardZ = Float.POSITIVE_INFINITY;

    public static boolean aimValid;
    public static Vec3 origin = Vec3.ZERO;
    public static Vec3 direction = new Vec3(0, 0, 1);
    /** Barrel ray hit a block or entity, so ringPoint is on that surface. */
    public static boolean hasSurface;
    /**
     * The barrel hit something the camera ray does not — a wall edge, a teammate, anything
     * between the muzzle and the camera's target.
     */
    public static boolean blocked;
    public static Vec3 ringPoint = Vec3.ZERO;
    public static boolean projectOk;
    public static float ndcX;
    public static float ndcY;

    private BarrelAim() {}

    /** Opens a new pass so each first-person draw picks the furthest muzzle bone again. */
    public static void beginCaptureFrame() {
        bestForwardZ = Float.POSITIVE_INFINITY;
    }

    /**
     * Stores the muzzle bone in the same camera-local space the laser uses.
     * Called while the first-person gun is being drawn, once the bone pose is current.
     */
    public static void capture(PoseStack.Pose bonePose) {
        float fov = (float) RenderEvents.currentFov;
        float fovScale = (float) (Math.tan(Math.toRadians(fov * 0.5)) / Math.tan(Math.toRadians(35.0)));
        if (!Float.isFinite(fovScale) || fovScale < 1.0e-3f) {
            fovScale = 1f;
        }
        Vector3f pos = bonePose.pose().getTranslation(new Vector3f());
        Vector3f dir = bonePose.pose().transformDirection(new Vector3f(0f, 0f, -1f));
        if (dir.lengthSquared() < 1.0e-8f) {
            return;
        }
        dir.normalize();
        dir.mul(fovScale, fovScale, 1f);
        // Viewmodel units become camera-space blocks via 0.25 * fov.
        // If a shader mod already scales the pose stack by 0.25, do not apply that factor twice.
        float unit = Client.isUsingIrisShader ? fovScale : 0.25f * fovScale;
        pos.mul(unit, unit, unit);
        if (pos.z > bestForwardZ) {
            return;
        }
        bestForwardZ = pos.z;
        localPos.set(pos);
        localDir.set(dir);
        captured = true;
        capturedAt = System.nanoTime();
    }

    public static void solve() {
        aimValid = false;
        hasSurface = false;
        blocked = false;
        projectOk = false;

        Minecraft mc = Minecraft.getInstance();
        if (mc.player == null || mc.level == null || mc.gameRenderer == null) {
            return;
        }
        if (mc.options.getCameraType() != CameraType.FIRST_PERSON) {
            return;
        }
        if (!Client.WEAPON_STATUS.isHoldingGun()) {
            return;
        }
        if (!captured || System.nanoTime() - capturedAt > STALE_NS) {
            return;
        }

        Camera camera = mc.gameRenderer.getMainCamera();
        Vector3f worldOff = new Vector3f(localPos).rotate(camera.rotation());
        Vector3f worldFwd = new Vector3f(localDir).rotate(camera.rotation());
        if (worldFwd.lengthSquared() < 1.0e-6f) {
            return;
        }
        worldFwd.normalize();

        Vec3 camPos = camera.getPosition();
        Vec3 muzzle = camPos.add(worldOff.x, worldOff.y, worldOff.z);
        if (!Double.isFinite(muzzle.x) || !Double.isFinite(muzzle.y) || !Double.isFinite(muzzle.z)) {
            return;
        }
        Vec3 forward = new Vec3(worldFwd.x, worldFwd.y, worldFwd.z);
        if (forward.lengthSqr() < 1.0e-6 || !Double.isFinite(forward.x) || !Double.isFinite(forward.y) || !Double.isFinite(forward.z)) {
            return;
        }
        forward = forward.normalize();

        Vec3 barrelHit = firstHit(mc.level, mc.player, muzzle, muzzle.add(forward.scale(BLOCK_RANGE)), ENTITY_RANGE);
        if (barrelHit != null) {
            hasSurface = true;
            ringPoint = barrelHit;
            Vec3 look = Vec3.directionFromRotation(camera.getXRot(), camera.getYRot());
            Vec3 lookHit = firstHit(mc.level, mc.player, camPos, camPos.add(look.scale(BLOCK_RANGE)), ENTITY_RANGE);
            double barrelDist = muzzle.distanceTo(barrelHit);
            double lookDist = lookHit == null ? BLOCK_RANGE : camPos.distanceTo(lookHit);
            blocked = barrelDist + 0.4 < lookDist;
        } else {
            ringPoint = muzzle.add(forward.scale(SKY_RANGE));
        }

        origin = muzzle;
        direction = forward;
        aimValid = true;
    }

    /** Projects ringPoint to NDC. Hidden when the point is behind the camera. */
    public static void project(Matrix4f modelView, Matrix4f projection, Vec3 camPos) {
        projectOk = false;
        if (!aimValid) {
            return;
        }
        Vector4f p = new Vector4f(
                (float) (ringPoint.x - camPos.x),
                (float) (ringPoint.y - camPos.y),
                (float) (ringPoint.z - camPos.z),
                1f);
        modelView.transform(p);
        projection.transform(p);
        if (p.w() <= 0.05f) {
            return;
        }
        float x = p.x() / p.w();
        float y = p.y() / p.w();
        if (!Float.isFinite(x) || !Float.isFinite(y)) {
            return;
        }
        ndcX = x;
        ndcY = y;
        projectOk = true;
    }

    public static void apply(GunFirePacket packet) {
        solve();
        if (aimValid) {
            packet.setBarrelAim(origin, direction);
        }
    }

    @Nullable
    private static Vec3 firstHit(Level level, Entity shooter, Vec3 start, Vec3 end, double entityRange) {
        Vec3 delta = end.subtract(start);
        double segLen = delta.length();
        if (segLen < 1.0e-4) {
            return null;
        }
        BlockHitResult blockHit = level.clip(new ClipContext(
                start, end, ClipContext.Block.COLLIDER, ClipContext.Fluid.NONE, shooter) {
            @Override
            public VoxelShape getBlockShape(BlockState state, BlockGetter blockGetter, BlockPos pos) {
                if (state.is(BlockTags.LEAVES)) {
                    return Shapes.empty();
                }
                return super.getBlockShape(state, blockGetter, pos);
            }
        });
        Vec3 blockPos = blockHit.getType() == HitResult.Type.MISS ? null : blockHit.getLocation();
        double bestDist2 = blockPos != null ? start.distanceToSqr(blockPos) : Double.POSITIVE_INFINITY;

        double entityLimit = Math.min(Math.max(entityRange, 0.0), segLen);
        Vec3 entityEnd = start.add(delta.scale(entityLimit / segLen));
        if (blockPos != null && bestDist2 <= entityLimit * entityLimit) {
            entityEnd = blockPos;
        }
        Vec3 best = blockPos;
        AABB box = new AABB(start, entityEnd).inflate(1.0);
        for (Entity entity : level.getEntities(shooter, box, e -> !e.isSpectator() && e.isPickable() && e.isAlive())) {
            var hit = entity.getBoundingBox().inflate(entity.getPickRadius()).clip(start, entityEnd);
            if (hit.isPresent()) {
                double dist2 = start.distanceToSqr(hit.get());
                if (dist2 < bestDist2) {
                    bestDist2 = dist2;
                    best = hit.get();
                }
            }
        }
        return best;
    }
}


2. New file: HUD reticle
------------------------
Save as your own client-only class. Call init() during client setup so the
static world task is registered. solve() and project() must run after the
level render, using the same model-view and projection matrices as any
first-person laser. In this mod those are LaserEffectRenderer.modelViewMat
and LaserEffectRenderer.projectionMat.

The world task here uses this mod's delayed Stage.HIGH. Any "after level,
before HUD" hook is enough. Do not register the class on the event bus
unless it has @SubscribeEvent methods.

drawHud() is called from the vanilla crosshair GUI layer, after that layer
is cancelled while a gun is held.

Replace Client.WEAPON_STATUS, IGun.getSpread, and the aiming-progress check
with the other mod's equivalents. getAimingProgress() <= 0.45 means the
crosshair is visible. Above that, every ADS form hides it, including irons.

package com.sheridan.gcr.client.render.fx;

import com.sheridan.gcr.Client;
import com.sheridan.gcr.client.BarrelAim;
import com.sheridan.gcr.client.WeaponStatus;
import com.sheridan.gcr.client.render.delayed.Stage;
import com.sheridan.gcr.client.render.delayed.Task;
import com.sheridan.gcr.modularSys.modules.guns.IGun;
import net.minecraft.client.Camera;
import net.minecraft.client.CameraType;
import net.minecraft.client.Minecraft;
import net.minecraft.client.gui.GuiGraphics;
import net.minecraft.util.Mth;
import net.minecraft.world.item.ItemStack;
import net.neoforged.api.distmarker.Dist;

/**
 * Camera dot at screen center, spread bars around it, and a hollow ring projected from the barrel.
 * The ring is where the shot will go. Aiming down sights hides it, including iron sights.
 */
@OnlyIn(Dist.CLIENT)
public final class BarrelCrosshairRenderer {
    /** Radius in GUI pixels. The stroke stays 1 screen pixel. */
    private static final int RING_RADIUS = 3;
    private static final int MARK = 0xFFFFFFFF;
    private static final int MARK_BLOCKED = 0xFFFFCC66;

    static {
        Stage.HIGH.addTask(new Task(event -> renderWorld()).forever());
    }

    private BarrelCrosshairRenderer() {}

    /** Loads the class so the world-render task in the static block is registered. */
    public static void init() {}

    private static void renderWorld() {
        BarrelAim.solve();
        if (!BarrelAim.aimValid || !shouldDraw()) {
            return;
        }
        Minecraft mc = Minecraft.getInstance();
        Camera camera = mc.gameRenderer.getMainCamera();
        BarrelAim.project(LaserEffectRenderer.modelViewMat, LaserEffectRenderer.projectionMat, camera.getPosition());
    }

    public static void drawHud(GuiGraphics graphics) {
        if (!shouldDraw()) {
            return;
        }
        var window = Minecraft.getInstance().getWindow();
        float guiScale = (float) window.getGuiScale();
        int screenW = window.getWidth();
        int screenH = window.getHeight();
        graphics.pose().pushPose();
        graphics.pose().scale(1f / guiScale, 1f / guiScale, 1f);

        int px = Math.max(1, Math.round(guiScale));
        int cx = screenW / 2;
        int cy = screenH / 2;
        drawCenter(graphics, cx, cy, px);
        if (BarrelAim.projectOk) {
            float sx = (BarrelAim.ndcX * 0.5f + 0.5f) * screenW;
            float sy = (1f - (BarrelAim.ndcY * 0.5f + 0.5f)) * screenH;
            int radius = RING_RADIUS * px;
            int margin = radius + 2;
            int x = Mth.clamp(Math.round(sx), margin, screenW - margin);
            int y = Mth.clamp(Math.round(sy), margin, screenH - margin);
            drawRing(graphics, x, y, radius, BarrelAim.blocked ? MARK_BLOCKED : MARK);
        }
        graphics.pose().popPose();
    }

    private static void drawCenter(GuiGraphics graphics, int cx, int cy, int px) {
        float spread = 0f;
        WeaponStatus status = Client.WEAPON_STATUS;
        IGun gun = status.getGun();
        ItemStack stack = status.getItemStack();
        if (gun != null && stack != null && !stack.isEmpty()) {
            spread = Math.max(0f, gun.getSpread(stack));
        }
        int gap = 2 * px + Math.round(spread * 6f * px);
        int arm = 2 * px;
        int y0 = cy - px / 2;
        int y1 = y0 + px;
        graphics.fill(cx - gap - arm, y0, cx - gap, y1, MARK);
        graphics.fill(cx + gap, y0, cx + gap + arm, y1, MARK);
        int d0 = cx - px / 2;
        int d1 = cy - px / 2;
        graphics.fill(d0, d1, d0 + px, d1 + px, MARK);
    }

    private static void drawRing(GuiGraphics graphics, int cx, int cy, int radius, int color) {
        for (int dy = -radius; dy <= radius; dy++) {
            for (int dx = -radius; dx <= radius; dx++) {
                if (Math.abs(Mth.sqrt(dx * dx + dy * dy) - radius) <= 0.5f) {
                    graphics.fill(cx + dx, cy + dy, cx + dx + 1, cy + dy + 1, color);
                }
            }
        }
    }

    private static boolean shouldDraw() {
        Minecraft mc = Minecraft.getInstance();
        if (mc.player == null || mc.screen != null || mc.options.hideGui) {
            return false;
        }
        if (mc.options.getCameraType() != CameraType.FIRST_PERSON) {
            return false;
        }
        WeaponStatus status = Client.WEAPON_STATUS;
        if (!status.isHoldingGun()) {
            return false;
        }
        return status.getAimingProgress() <= 0.45f;
    }
}


3. Hook: client setup
---------------------
Call this once on the client so the renderer's static block runs.

    BarrelCrosshairRenderer.init();


4. Hook: start of first-person gun render
-----------------------------------------
At the start of the local player's first-person gun draw, after any early
return for shadows or a hidden viewmodel:

    BarrelAim.beginCaptureFrame();


5. Hook: muzzle bone, during that same render
---------------------------------------------
Call capture only for the local player, first person, and the gun root.
Skip the bone when a child occupies the muzzle slot; the child's bone is the
real tip. Bone name in this mod is "MUZZLE_FLASH". Slot name is "MUZZLE".
Forward in viewmodel space is -Z. The pose is only valid during model.render.

    if (context.isFirstPerson()
            && context.root != null
            && effectModuleId.equals(context.root.id)
            && context.entity.getId() == Client.LOCAL_PLAYER_ID) {
        BarrelAim.capture(bonePose);
    }

Capture even when the gun is not currently firing. Do it before any
"no muzzle flash this frame" return.


6. Hook: cancel the vanilla crosshair and draw the reticle
----------------------------------------------------------
On RenderGuiLayerEvent.Pre for VanillaGuiLayers.CROSSHAIR, when the player
is holding a gun:

    event.setCanceled(true);
    BarrelCrosshairRenderer.drawHud(event.getGuiGraphics());


7. Hook: client fire packet
---------------------------
Add these fields:

    public boolean useBarrel;
    public double originX;
    public double originY;
    public double originZ;
    public float dirX;
    public float dirY;
    public float dirZ;

    public void setBarrelAim(Vec3 origin, Vec3 direction) {
        this.useBarrel = true;
        this.originX = origin.x;
        this.originY = origin.y;
        this.originZ = origin.z;
        this.dirX = (float) direction.x;
        this.dirY = (float) direction.y;
        this.dirZ = (float) direction.z;
    }

Write the boolean, then the six numbers only when it is true. Read them the
same way.

Right before sending the fire packet to the server:

    BarrelAim.apply(packet);


8. Hook: server spawn
---------------------
Spread is still applied. Gun kick is not added again when the barrel aim is
accepted, because the viewmodel pose already includes it.

    float spread = getSpread(itemStack);
    float randPitch = (float) ((random.nextGaussian() - 0.5f) * spread);
    float randYaw = (float) ((random.nextGaussian() - 0.5f) * spread);

    Vec3 pos;
    Vec3 dir;
    if (shooter instanceof ServerPlayer player && acceptBarrelAim(player, packet)) {
        pos = new Vec3(packet.originX, packet.originY, packet.originZ);
        dir = deflect(new Vec3(packet.dirX, packet.dirY, packet.dirZ), randPitch, randYaw);
    } else {
        float yaw = shooter.getYRot() + packet.gunKickYaw + randYaw;
        float pitch = shooter.getXRot() + packet.gunKickPitch + randPitch;
        dir = Vec3.directionFromRotation(pitch, yaw).normalize();
        pos = shooter.getEyePosition();
    }

    private static boolean acceptBarrelAim(ServerPlayer player, GunFirePacket packet) {
        if (!packet.useBarrel) {
            return false;
        }
        if (!Double.isFinite(packet.originX) || !Double.isFinite(packet.originY) || !Double.isFinite(packet.originZ)) {
            return false;
        }
        if (!Float.isFinite(packet.dirX) || !Float.isFinite(packet.dirY) || !Float.isFinite(packet.dirZ)) {
            return false;
        }
        Vec3 origin = new Vec3(packet.originX, packet.originY, packet.originZ);
        if (origin.distanceToSqr(player.getEyePosition()) > 25.0) {
            return false;
        }
        double len2 = packet.dirX * (double) packet.dirX + packet.dirY * (double) packet.dirY + packet.dirZ * (double) packet.dirZ;
        if (len2 < 0.25 || len2 > 4.0) {
            return false;
        }
        Vec3 dir = new Vec3(packet.dirX, packet.dirY, packet.dirZ).normalize();
        return dir.dot(player.getLookAngle()) >= 0.0;
    }

    private static Vec3 deflect(Vec3 dir, float pitchDeg, float yawDeg) {
        dir = dir.normalize();
        float yaw = (float) Math.toDegrees(Mth.atan2(-dir.x, dir.z));
        float pitch = (float) Math.toDegrees(Math.asin(Mth.clamp(-dir.y, -1.0, 1.0)));
        return Vec3.directionFromRotation(pitch + pitchDeg, yaw + yawDeg);
    }

Spawn the projectile at pos with velocity dir * speed. This mod uses 30 blocks
per tick. Match the bullet's block filter to firstHit(), including ignoring
leaves, or the ring will mark surfaces the bullet flies through.


9. What you still have to supply in the other mod
--------------------------------------------------
- A first-person muzzle matrix every frame the gun is drawn.
- The current FOV, and whether a shader pack already scaled the viewmodel by 0.25.
- Model-view and projection matrices from the level render, for projecting the ring.
- A client-to-server fire packet.
- An aim-progress value in the 0..1 range if ADS should hide the HUD.
- The same collision rules the projectile uses.

The ring is one frame behind the viewmodel, because the hand is drawn after
the world stage that projects it. Third person, and a missing bone, fall back
to the eyes and show only the center dot.
